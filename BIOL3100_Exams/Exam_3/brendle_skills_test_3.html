##Task 5 - GLM. I fit separate generalized linear models for each chemical, using Concentration as the response and YearsSinceBurn as the predictor. I used broom::tidy() to extract coefficients and then filtered for the YearsSinceBurn term with p < 0.05. The resulting table shows which chemical concentrations are significantly affected by time since burn.

library(broom)
library(dplyr)

# Get list of all compounds
compounds <- unique(juniper_tidy$Compound)

# Fit a glm for each compound and store results
res_list <- lapply(compounds, function(cmp) {
  dat <- juniper_tidy %>% filter(Compound == cmp)
  
  # fit GLM (gaussian family ~ same as linear model)
  m <- glm(Concentration ~ YearsSinceBurn, data = dat, family = gaussian())
  
  # tidy the model output and add the compound name as a column
  tidy(m) %>%
    mutate(Compound = cmp)
})

# Combine all results into one data frame
glm_results <- bind_rows(res_list)

# Keep only significant YearsSinceBurn terms
glm_sig <- glm_results %>%
  filter(term == "YearsSinceBurn",
         p.value < 0.05) %>%
  arrange(p.value)

glm_sig
