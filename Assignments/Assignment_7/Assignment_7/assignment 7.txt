library(readr)
library(janitor)

dat <- read_csv("Utah_Religions_by_County.csv", show_col_types = FALSE) |>
  clean_names()

names(dat)

group_cols <- setdiff(names(dat), c("county", "pop_2010"))
relig_long <- dat %>%
  pivot_longer(
    cols = all_of(group_cols),
    names_to = "group",
    values_to = "adherents"
  ) %>%
  mutate(
    prop = adherents / pop_2010
  )
nonrel_by_county <- dat %>%
  transmute(
    county,
    non_rel_prop = non_religious / pop_2010
  )
These lines kept the pop values as whole numbers but now the rest of the stats are as proportions/percentages.

relig_long <- relig_long %>%
  left_join(nonrel_by_county, by = "county")
cor_pop_vs_prop <- relig_long %>%
  group_by(group) %>%
  summarize(
    r = cor(pop_2010, prop, use = "pairwise.complete.obs"),
    .groups = "drop"
  ) %>%
  arrange(desc(abs(r)))

cor_pop_vs_prop

ggplot(relig_long, aes(x = pop_2010, y = prop)) +
  geom_point(alpha = 0.6) +
  geom_smooth(method = "lm", se = FALSE, linewidth = 0.7, color = "steelblue") +
  facet_wrap(~ group, scales = "free_y") +
  scale_y_continuous(labels = percent_format(accuracy = 1)) +
  scale_x_continuous(labels = label_number(big.mark = ",")) +
  labs(
    title = "Population vs Proportion of Each Religious Group (by County)",
    x = "County population (2010 census)",
    y = "Proportion of county"
  ) +
  theme_bw()
